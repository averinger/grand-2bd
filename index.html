/* ═══ ЗАГРУЗКА МОДЕЛИ с честным fallback ═══ */ async function loadModel() { startBtn.disabled = true; startBtn.textContent = 'Загрузка...'; progressBox.classList.add('show'); progressFill.style.width = '0%'; progressText.textContent = 'Инициализация...'; sbDot.className = 'dot loading'; sbStatus.textContent = 'Инициализация...'; const model = state.settings.model; lsModel.textContent = model.split('/').pop().replace('-Instruct', ''); const progressCb = (p) => { if (p.status === 'progress') { const pct = Math.round(p.progress || 0); progressFill.style.width = pct + '%'; const mb = ((p.loaded || 0) / 1048576).toFixed(1); const total = ((p.total || 0) / 1048576).toFixed(1); progressText.textContent = `${p.file || 'файл'} — ${mb} / ${total} МБ (${pct}%)`; } else if (p.status === 'ready') { progressFill.style.width = '100%'; progressText.textContent = 'Готово!'; } else if (p.status === 'initiate') { progressText.textContent = `Скачивание ${p.file || '...'}`; } }; let generator = null; let device = 'wasm'; // ── Шаг 1: пробуем WebGPU по-настоящему ── if (navigator.gpu) { try { progressText.textContent = 'Пробуем WebGPU...'; const adapter = await navigator.gpu.requestAdapter(); if (adapter) { generator = await pipeline('text-generation', model, { device: 'webgpu', dtype: 'q4', progress_callback: progressCb }); device = 'webgpu'; } else { console.log('adapter = null → пропускаем WebGPU'); } } catch (e) { console.warn('WebGPU упал, откатываемся на WASM:', e); generator = null; progressFill.style.width = '0%'; } } // ── Шаг 2: если WebGPU не взлетел — грузим WASM ── if (!generator) { try { progressText.textContent = 'Загрузка на CPU (WASM)...'; generator = await pipeline('text-generation', model, { device: 'wasm', dtype: 'q4', progress_callback: progressCb }); device = 'wasm'; } catch (err) { console.error('Фатальная ошибка загрузки:', err); startBtn.disabled = false; startBtn.textContent = 'Повторить'; progressText.textContent = 'Ошибка: ' + err.message; sbDot.className = 'dot'; sbStatus.textContent = 'Ошибка'; return; } } // ── Успех ── state.generator = generator; state.device = device; lsDevice.textContent = device === 'webgpu' ? 'WebGPU (GPU)' : 'WASM (CPU)'; sbDot.className = 'dot ready'; sbStatus.textContent = device === 'webgpu' ? 'Готов (GPU)' : 'Готов (CPU)'; progressFill.style.width = '100%'; progressText.textContent = 'Модель загружена!'; setTimeout(() => { loader.classList.add('hide'); inputEl.disabled = false; sendBtn.disabled = false; inputEl.focus(); }, 600); }
